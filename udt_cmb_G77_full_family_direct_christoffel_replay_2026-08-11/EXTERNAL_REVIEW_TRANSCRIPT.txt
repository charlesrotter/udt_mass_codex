   303	def solid_angle(a: np.ndarray, b: np.ndarray, c: np.ndarray) -> np.ndarray:
   304	    numerator = np.einsum("bi,bi->b", a, np.cross(b, c))
   305	    denominator = (
   306	        1.0
   307	        + np.einsum("bi,bi->b", a, b)
   308	        + np.einsum("bi,bi->b", b, c)
   309	        + np.einsum("bi,bi->b", c, a)
   310	    )
   311	    return 2.0 * np.arctan2(numerator, denominator)

   314	def projected_face_orientation(vertices: np.ndarray, faces: np.ndarray, endpoint: np.ndarray, valid: np.ndarray) -> np.ndarray:
   315	    output = np.full(len(faces), np.nan)
   316	    selected = np.flatnonzero(valid)
   317	    if not len(selected):
   318	        return output
   319	    tri_in = vertices[faces[selected]]
   320	    tri_out = endpoint[faces[selected]]
   321	    center_in = np.sum(tri_in, axis=1)
   322	    center_out = np.sum(tri_out, axis=1)
   323	    center_in /= np.linalg.norm(center_in, axis=1)[:, None]
   324	    center_out /= np.linalg.norm(center_out, axis=1)[:, None]
   325	    edge_in_1 = tri_in[:, 1] - tri_in[:, 0]
   326	    edge_in_2 = tri_in[:, 2] - tri_in[:, 0]
   327	    edge_out_1 = tri_out[:, 1] - tri_out[:, 0]
   328	    edge_out_2 = tri_out[:, 2] - tri_out[:, 0]
   329	    edge_in_1 -= np.einsum("bi,bi->b", edge_in_1, center_in)[:, None] * center_in
   330	    edge_in_2 -= np.einsum("bi,bi->b", edge_in_2, center_in)[:, None] * center_in
   331	    edge_out_1 -= np.einsum("bi,bi->b", edge_out_1, center_out)[:, None] * center_out
   332	    edge_out_2 -= np.einsum("bi,bi->b", edge_out_2, center_out)[:, None] * center_out
   333	    det_in = np.einsum("bi,bi->b", center_in, np.cross(edge_in_1, edge_in_2))
   334	    det_out = np.einsum("bi,bi->b", center_out, np.cross(edge_out_1, edge_out_2))
   335	    usable = np.abs(det_in) > 1.0e-14
   336	    output[selected[usable]] = det_out[usable] / det_in[usable]
   337	    return output
