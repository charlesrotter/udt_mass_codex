   257	def shortest_count(kind: str) -> tuple[int, list[list[int]]]:
   258	    values = {}
   259	    for p in range(-4, 5):
   260	        for q in range(-4, 5):
   261	            if (p, q) == (0, 0) or math.gcd(p, q) != 1:
   262	                continue
   263	            w = primitive_line(p, q)
   264	            values[w] = p * p + q * q if kind == "round" else p * p - p * q + q * q
   265	    minimum = min(values.values())
   266	    winners = sorted([list(w) for w, value in values.items() if value == minimum])
   267	    return len(winners), winners

   270	def multiply_polynomials(left: list[int], right: list[int]) -> list[int]:
   271	    result = [0] * (len(left) + len(right) - 1)
   272	    for i, a in enumerate(left):
